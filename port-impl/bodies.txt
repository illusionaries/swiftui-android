# OptionSet structs store `rawValue`, so their raw-value initializer is a field assignment. The
# declaration must stay (dropping it changes how Swift synthesises the RawRepresentable witness).
SwiftUICore.Edge.Set.init(rawValue:) => self.rawValue = rawValue
SwiftUICore.HorizontalDirection.Set.init(rawValue:) => self.rawValue = rawValue
SwiftUICore.VerticalDirection.Set.init(rawValue:) => self.rawValue = rawValue
