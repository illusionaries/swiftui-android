# Stub bodies the engine supplies in place (symbol => Swift expression).
#
# Omitted declarations live in port-impl/implemented.txt; this file is for stubs that must stay
# declared and simply need a trivial body.

# OptionSet structs store `rawValue`, so their raw-value initializer is a field assignment. The
# declaration must stay (dropping it changes how Swift synthesises the RawRepresentable witness).
SwiftUICore.Edge.Set.init(rawValue:) => self.rawValue = rawValue
SwiftUICore.HorizontalDirection.Set.init(rawValue:) => self.rawValue = rawValue
SwiftUICore.VerticalDirection.Set.init(rawValue:) => self.rawValue = rawValue

# Structs whose initializer is synthesised by the compiler: keep the declaration, provide the body.
SwiftUICore.Gradient.init(stops:) => self.stops = stops
SwiftUICore.Gradient.Stop.init(color:location:) => self.color = color; self.location = location
SwiftUICore.Gradient.Stop.==(a:b:) => PortStyleRegistry.pointer(of: a.color) == PortStyleRegistry.pointer(of: b.color) && a.location == b.location
SwiftUICore.LinearGradient.init(gradient:startPoint:endPoint:) => self.gradient = gradient; self.startPoint = startPoint; self.endPoint = endPoint

# Layout modifier structs: their fields are real storage in the interface; the initialisers and
# animatable data are what Apple's binary supplied.
SwiftUICore._FrameLayout.init(width:height:alignment:) => self.width = width; self.height = height; self.alignment = alignment
SwiftUICore._FrameLayout.animatableData => fatalError("animatableData is not used by the port")
SwiftUICore._FlexFrameLayout.init(minWidth:idealWidth:maxWidth:minHeight:idealHeight:maxHeight:alignment:) => self.minWidth = minWidth; self.idealWidth = idealWidth; self.maxWidth = maxWidth; self.minHeight = minHeight; self.idealHeight = idealHeight; self.maxHeight = maxHeight; self.alignment = alignment
SwiftUICore._FlexFrameLayout.animatableData => fatalError("animatableData is not used by the port")
SwiftUICore._PaddingLayout.animatableData => fatalError("animatableData is not used by the port")

# LocalizedStringKey keeps its text in real storage; `arguments` is private, so the assignments
# must live in the generated file itself (this override is applied there).
SwiftUICore.LocalizedStringKey.init(_:) => self.key = value; self.hasFormatting = false; self.arguments = []
SwiftUICore.LocalizedStringKey.init(stringLiteral:) => self.key = value; self.hasFormatting = false; self.arguments = []

# LocalizedStringKey interpolation accumulates plain text (Android has no resource bundles yet).
# The occurrence selectors target overloads that share one symbol string.
SwiftUICore.LocalizedStringKey.StringInterpolation.init(literalCapacity:interpolationCount:) => self._portText = ""
SwiftUICore.LocalizedStringKey.StringInterpolation.appendLiteral(_:) => self._portText += literal
SwiftUICore.LocalizedStringKey.StringInterpolation.appendInterpolation(_:)#1 => self._portText += string
SwiftUICore.LocalizedStringKey.StringInterpolation.appendInterpolation(_:specifier:) => self._portText += String(describing: value)
SwiftUICore.LocalizedStringKey.init(stringInterpolation:) => self.key = stringInterpolation._portText; self.hasFormatting = false; self.arguments = []

# Raw values of Apple's small enums (declaration order is the raw value).
SwiftUICore.Axis.rawValue => switch self { case .horizontal: return 0; case .vertical: return 1 }

# `ForEach`'s public initialisers funnel through these internal factories; the struct's own
# storage (`data`, `content`) is real, so the factory just builds the value.
SwiftUICore.ForEach.create(_:?:) => ForEach(data: data, content: content)
SwiftUICore.ForEach.create(_:id:?:) => ForEach(data: data, content: content)

# GridItem stores its own size/spacing/alignment; only the initialiser was Apple's.
SwiftUICore.GridItem.init(_:spacing:alignment:) => self.size = size; self.spacing = spacing; self.alignment = alignment

# ObservedObject keeps its object in the stored `wrappedValue` field; only the seed needs setting.
SwiftUICore.ObservedObject.init(wrappedValue:) => self.wrappedValue = wrappedValue; self._seed = 0

# Property-wrapper accessors stay declared (Swift requires `wrappedValue` on the type before the
# type can be used as an attribute), so their bodies are supplied here: #1 is the getter, #2 the
# setter, in declaration order.
SwiftUICore.Binding.wrappedValue#1 => (location as? any PortLocationBox<Value>)?.value ?? _value
SwiftUICore.Binding.wrappedValue#2 => (location as? any PortLocationBox<Value>)?.value = newValue; PortStateStore.markDirty()
SwiftUICore.Binding.projectedValue => self
SwiftUICore.State.wrappedValue#1 => (_location as? any PortLocationBox<Value>)?.value ?? _value
SwiftUICore.State.wrappedValue#2 => (_location as? any PortLocationBox<Value>)?.value = newValue; PortStateStore.markDirty()
SwiftUICore.State.projectedValue => Binding(get: { self.wrappedValue }, set: { self.wrappedValue = $0 })
SwiftUICore.LazyState.wrappedValue#1 => (_location as? any PortLocationBox<Value>)?.value ?? { switch _storage { case let .value(value): return value; case let .thunk(thunk): return thunk() } }()
SwiftUICore.LazyState.wrappedValue#2 => (_location as? any PortLocationBox<Value>)?.value = newValue; PortStateStore.markDirty()
SwiftUICore.LazyState.projectedValue => Binding(get: { self.wrappedValue }, set: { self.wrappedValue = $0 })
SwiftUICore.ObservedObject.projectedValue => Wrapper(root: wrappedValue)
SwiftUICore.StateObject.wrappedValue => switch storage { case let .object(observed): return observed.wrappedValue; case let .initially(thunk): return PortObjectStore.object(at: _portSlot, create: thunk) }
SwiftUICore.StateObject.projectedValue => ObservedObject(wrappedValue: wrappedValue).projectedValue
SwiftUICore.PinnedScrollableViews.init(rawValue:) => self.rawValue = rawValue
